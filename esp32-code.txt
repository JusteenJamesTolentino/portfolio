#include <WiFi.h>
#include <PubSubClient.h>

const char* mqtt_server = "broker.hivemq.com";

WiFiClient espClient;
PubSubClient client(espClient);

String deviceId = "";
String ssid = "";
String password = "";

void setup_wifi() {
  Serial.println("STATUS:Connecting to Wi-Fi...");
  WiFi.begin(ssid.c_str(), password.c_str());

  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
    attempts++;
    // Timeout after 20 attempts (10 seconds)
    if (attempts > 20) {
      Serial.println("\nSTATUS:Wi-Fi connection failed. Please restart and check credentials.");
      return;
    }
  }
  Serial.println("\nSTATUS:WiFi connected successfully!");
  Serial.println("STATUS:IP Address: " + WiFi.localIP().toString());
}

void callback(char* topic, byte* payload, unsigned int length) {
  String message = "";
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  Serial.println("MSG:" + message);
}

void reconnect() {
  while (!client.connected()) {
    if (client.connect(deviceId.c_str())) {
      client.subscribe("cpe/messenger/global");
    } else {
      delay(5000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  // Generate unique ID based on MAC address
  uint64_t chipid = ESP.getEfuseMac();
  char idBuffer[23];
  sprintf(idBuffer, "ESP32_%04X%08X", (uint16_t)(chipid >> 32), (uint32_t)chipid);
  deviceId = String(idBuffer);

  // Send the hardware ID to the web app
  Serial.println("ID:" + deviceId);

  // --- INTERACTIVE WIFI PROMPT OVER SERIAL ---
  Serial.println("PROMPT:Enter WiFi SSID:");
  while (ssid.length() == 0) {
    if (Serial.available() > 0) {
      ssid = Serial.readStringUntil('\n');
      ssid.trim();
    }
  }
  Serial.println("Received SSID: " + ssid);

  Serial.println("PROMPT:Enter WiFi Password:");
  while (password.length() == 0) {
    if (Serial.available() > 0) {
      password = Serial.readStringUntil('\n');
      password.trim();
    }
  }
  Serial.println("Received Password. Connecting...");

  setup_wifi();
  client.setServer(mqtt_server, 1883);
  client.setCallback(callback);
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    if (!client.connected()) {
      reconnect();
    }
    client.loop();
  }

  // Read messages coming from the Web Browser via Serial
  if (Serial.available() > 0) {
    String input = Serial.readStringUntil('\n');
    input.trim();
    if (input.length() > 0) {
      String payload = deviceId + ":" + input;
      client.publish("cpe/messenger/global", payload.c_str());
    }
  }
}
